// App はアプリケーションの組み立てと起動を担当する。
type App struct {
        db    *gorm.DB
        redis *redis.Client
        e     *echo.Echo
}

// New は App を生成する。DI の入口。
func New() (*App, error) {
        cfg := db.NewConfig()
        gormDB, err := db.NewDB(cfg)
        if err != nil {
                return nil, err
        }

        log.Println("Successfully connected to PostgreSQL (GORM)")

        return &App{db: gormDB}, nil
}

// Run はアプリケーションを起動する。
func (a *App) Run() error {
        // Repository 生成（port を実装する GORM adapter）
        authCodeRepo := gormrepo.NewAuthCodeRepository(a.db)
        sessionRepo := gormrepo.NewSessionRepository(a.db)
        clientRepo := gormrepo.NewClientRepository(a.db)
        tokenRepo := gormrepo.NewTokenRepository(a.db)
        userRepo := gormrepo.NewUserRepository(a.db)

        // Redis（login_challenge 用）
        redisCfg := redisinfra.NewConfig()
        redisClient, err := redisinfra.NewClient(redisCfg)
        if err != nil {
                return fmt.Errorf("redis: %w", err)
        }
        a.redis = redisClient
        authReqRepo := redisinfra.NewAuthorizationRequestRepository(redisClient)

        // JWT Signer（RS256、開発用は起動時に鍵生成）
        jwtKey, err := jwt.GenerateKey()
        if err != nil {
                return fmt.Errorf("generate jwt key: %w", err)
        }
        jwtSigner := jwt.NewSigner(jwtKey)

        // TransactionRunner（code 交換の二重実行防止用）
        txRunner := db.NewTransactionRunner(a.db)

        // Issuer（ID Token の iss、Discovery、全 endpoint URL の起点。本番は https 必須）
        issuer := getEnv("OAUTH_ISSUER", "http://localhost:8080")

        // ログイン/認可URL（環境変数で上書き可能）
        loginURL := getEnv("LOGIN_URL", "http://localhost:5173/login")
        consentURL := getEnv("CONSENT_URL", "http://localhost:5173/consent")
        serverBaseURL := getEnv("SERVER_BASE_URL", "http://localhost:8080")

        // CORS: React（別オリジン）から credentials 付きで Cookie を受け取るために必須
        // AllowCredentials=true のとき AllowOrigins は * 不可。CORS_ORIGIN でフロントのオリジンを指定。
        corsOrigin := getEnv("CORS_ORIGIN", "http://localhost:5173")

        // Usecase 生成
        authorizeUseCase := authorize.NewUseCase(authCodeRepo, clientRepo, authReqRepo)
        sessionResolver := session.NewResolver(sessionRepo)
        tokenUseCase := token.NewUseCase(txRunner, authCodeRepo, tokenRepo, clientRepo, sessionRepo, jwtSigner, issuer)
        loginUseCase := login.NewUseCase(userRepo, sessionRepo, authReqRepo, serverBaseURL)
        logoutUseCase := logout.NewUseCase(sessionRepo)
        userInfoUseCase := userinfo.NewUseCase(tokenRepo, userRepo, sessionRepo, jwtSigner)

        // Handler 生成
        authorizeHandler := handler.NewAuthorizeHandler(authorizeUseCase, sessionResolver, loginURL, consentURL, serverBaseURL)
        tokenHandler := handler.NewTokenHandler(tokenUseCase, clientRepo, []string{corsOrigin})
        loginHandler := handler.NewLoginHandler(loginUseCase)
        logoutHandler := handler.NewLogoutHandler(logoutUseCase)
        healthHandler := handler.NewHealthHandler()
        discoveryHandler := handler.NewDiscoveryHandler(issuer)
        jwksHandler := handler.NewJWKSHandler(jwtSigner) // *jwt.Signer は port.JWKSProvider を実装
        userInfoHandler := handler.NewUserInfoHandler(userInfoUseCase)

        port := getEnv("SERVER_PORT", "8080")
        a.e = echo.New()

        // CORS: credentials 付きリクエストで Cookie を返すために必須
        a.e.Use(middleware.CORSWithConfig(middleware.CORSConfig{
                AllowOrigins:     []string{corsOrigin},
                AllowMethods:     []string{"GET", "POST", "OPTIONS"},
                AllowHeaders:     []string{"Content-Type", "Authorization"},
                AllowCredentials: true,
        }))

        router.New(a.e, &router.Config{
                AuthorizeHandler: authorizeHandler,
                TokenHandler:     tokenHandler,
                LoginHandler:     loginHandler,
                LogoutHandler:    logoutHandler,
                HealthHandler:    healthHandler,
                DiscoveryHandler: discoveryHandler,
                JWKSHandler:      jwksHandler,
                UserInfoHandler:  userInfoHandler,
        })

        // Graceful shutdown
        go func() {
                sigCh := make(chan os.Signal, 1)
                signal.Notify(sigCh, syscall.SIGINT, syscall.SIGTERM)
                <-sigCh
                log.Println("Shutting down...")
                if err := a.e.Shutdown(context.Background()); err != nil {
                        log.Printf("Shutdown error: %v", err)
                }
        }()

        addr := fmt.Sprintf(":%s", port)
        log.Printf("Server starting on %s", addr)
        return a.e.Start(addr)
}

func getEnv(key, defaultValue string) string {
        if v := os.Getenv(key); v != "" {
                return v
        }
        return defaultValue
}

                a.redis.Close()
        }
        if a.db != nil {

                        sqlDB.Close()
                }
        }

// sessionCookieName は /sessions/login でセットする Cookie 名（login_handler と共通）
const sessionCookieName = "oauth_session"

// sessionMaxAge はセッション Cookie の有効期限（秒）。login_handler で使用。
const sessionMaxAge = 24 * 60 * 60

// authorizeParams は GET の認可リクエストパラメータ。
type authorizeParams struct {
        ClientID            string
        RedirectURI         string
        ResponseType        string
        Scope               string
        State               string
        Nonce               string
        CodeChallenge       string
        CodeChallengeMethod string
        Prompt              string // login, consent, none
}

// SessionResolver はセッション解決のインターフェース（authorize は code 発行のみのため分離）。
type SessionResolver interface {
        Resolve(ctx context.Context, sessionToken string) (sessionID, userID uuid.UUID, ok bool, err error)
}

// AuthorizeHandler は認可エンドポイントのハンドラ。
// 責務：request validation、セッションCookie確認、なければ login へ redirect、あれば IssueCode。
type AuthorizeHandler struct {
        authorizeUseCase authorize.Interactor
        sessionResolver  SessionResolver
        loginURL         string
        consentURL       string
        serverBaseURL    string
}

// NewAuthorizeHandler は AuthorizeHandler を生成する。
func NewAuthorizeHandler(authorizeUseCase authorize.Interactor, sessionResolver SessionResolver, loginURL, consentURL, serverBaseURL string) *AuthorizeHandler {
        return &AuthorizeHandler{
                authorizeUseCase: authorizeUseCase,
                sessionResolver:  sessionResolver,
                loginURL:         loginURL,
                consentURL:       consentURL,
                serverBaseURL:    serverBaseURL,
        }
}

// Get は認可リクエストを処理する。
// ログイン済みなら code 発行→redirect。未ログインなら login_challenge 発行して React へリダイレクト。
// GET /oauth/authorize?client_id=...&redirect_uri=...&response_type=code&scope=...&state=...&code_challenge=...&code_challenge_method=S256
func (h *AuthorizeHandler) Get(c echo.Context) error {
        params := parseAuthorizeParams(c)
        if err := h.validateAuthorizeParams(c, params); err != nil {
                return err
        }

        sessionToken := ""
        if cookie, err := c.Cookie(sessionCookieName); err == nil && cookie.Value != "" {
                sessionToken = cookie.Value
        }
        sessionID, userID, ok, err := h.sessionResolver.Resolve(c.Request().Context(), sessionToken)
        if err != nil {
                return c.JSON(http.StatusInternalServerError, map[string]string{"error": "server_error"})
        }
        if ok && sessionID != uuid.Nil && userID != uuid.Nil {
                // ログイン済み → code 発行して redirect（approve フォームなし）
                clientID, _ := uuid.Parse(params.ClientID)
                scope := params.Scope
                if scope == "" {
                        scope = "openid"
                }
                out, err := h.authorizeUseCase.IssueCode(c.Request().Context(), &authorize.IssueCodeInput{
                        ResponseType:        "code",
                        UserID:              userID,
                        SessionID:           sessionID,
                        ClientID:            clientID,
                        RedirectURI:         params.RedirectURI,
                        Scopes:              scope,
                        State:               params.State,
                        Nonce:               params.Nonce,
                        CodeChallenge:       params.CodeChallenge,
                        CodeChallengeMethod: params.CodeChallengeMethod,
                })
                if err != nil {
                        return h.authorizeErrorResponse(c, err)
                }
                u, _ := url.Parse(params.RedirectURI)
                q := u.Query()
                q.Set("code", out.Code)
                if out.State != "" {
                        q.Set("state", out.State)
                }
                u.RawQuery = q.Encode()
                return c.Redirect(http.StatusFound, u.String())
        }

        // 未ログイン → login_challenge 発行して React へリダイレクト
        _, redirectURL, err := h.authorizeUseCase.CreateLoginChallenge(c.Request().Context(), &authorize.StoredAuthParams{
                ClientID:            params.ClientID,
                RedirectURI:         params.RedirectURI,
                ResponseType:        params.ResponseType,
                Scope:               params.Scope,
                State:               params.State,
                Nonce:               params.Nonce,
                CodeChallenge:       params.CodeChallenge,
                CodeChallengeMethod: params.CodeChallengeMethod,
                Prompt:              params.Prompt,
                LoginURL:            h.loginURL,
        })
        if err != nil {
                return c.JSON(http.StatusInternalServerError, map[string]string{"error": "server_error"})
        }
        return c.Redirect(http.StatusFound, redirectURL)
}

// GetContinue はログイン後の認可リクエスト再開を処理する。
// GET /oauth/authorize/continue?login_challenge=...
func (h *AuthorizeHandler) GetContinue(c echo.Context) error {
        challenge := c.QueryParam("login_challenge")
        if challenge == "" {
                return c.JSON(http.StatusBadRequest, map[string]string{
                        "error":             "invalid_request",
                        "error_description": "login_challenge is required",
                })
        }

        sessionToken := ""
        if cookie, err := c.Cookie(sessionCookieName); err == nil && cookie.Value != "" {
                sessionToken = cookie.Value
        }
        sessionID, userID, ok, err := h.sessionResolver.Resolve(c.Request().Context(), sessionToken)
        if err != nil {
                return c.JSON(http.StatusInternalServerError, map[string]string{"error": "server_error"})
        }
        if !ok || sessionID == uuid.Nil || userID == uuid.Nil {
                redirectURL := h.loginURL + "?login_challenge=" + challenge
                return c.Redirect(http.StatusFound, redirectURL)
        }

        // prompt=consent かつ approved でない場合は同意画面へ（消費せず）
        approved := c.QueryParam("approved") == "true"
        if !approved {
                storedPeek, err := h.authorizeUseCase.GetChallenge(c.Request().Context(), challenge)
                if err == nil && storedPeek != nil && storedPeek.Prompt == "consent" && h.consentURL != "" {
                        redirectURL := h.consentURL + "?login_challenge=" + challenge + "&scope=" + url.QueryEscape(storedPeek.Scope)
                        return c.Redirect(http.StatusFound, redirectURL)
                }
        }

        // 消費して code 発行
        stored, err := h.authorizeUseCase.GetAndConsumeChallenge(c.Request().Context(), challenge)
        if err != nil {
                if errors.Is(err, authorize.ErrChallengeNotFound) {
                        return c.JSON(http.StatusBadRequest, map[string]string{
                                "error":             "invalid_request",
                                "error_description": "login challenge not found or expired",
                        })
                }
                return c.JSON(http.StatusInternalServerError, map[string]string{"error": "server_error"})
        }

        clientID, _ := uuid.Parse(stored.ClientID)
        scope := stored.Scope
        if scope == "" {
                scope = "openid"
        }

        out, err := h.authorizeUseCase.IssueCode(c.Request().Context(), &authorize.IssueCodeInput{
                ResponseType:        "code",
                UserID:             userID,
                SessionID:          sessionID,
                ClientID:           clientID,
                RedirectURI:        stored.RedirectURI,
                Scopes:             scope,
                State:              stored.State,
                Nonce:              stored.Nonce,
                CodeChallenge:     stored.CodeChallenge,
                CodeChallengeMethod: stored.CodeChallengeMethod,
        })
        if err != nil {
                return h.authorizeErrorResponse(c, err)
        }

        u, _ := url.Parse(stored.RedirectURI)
        q := u.Query()
        q.Set("code", out.Code)
        if out.State != "" {
                q.Set("state", out.State)
        }
        u.RawQuery = q.Encode()

        return c.Redirect(http.StatusFound, u.String())
}

// parseBasicAuth は Authorization: Basic base64(client_id:client_secret) をパースする。
func parseBasicAuth(auth string) (clientID, clientSecret string, ok bool) {
        if auth == "" || !strings.HasPrefix(auth, "Basic ") {
                return "", "", false
        }
        decoded, err := base64.StdEncoding.DecodeString(strings.TrimPrefix(auth, "Basic "))
        if err != nil {
                return "", "", false
        }
        parts := strings.SplitN(string(decoded), ":", 2)
        if len(parts) != 2 {
                return "", "", false
        }
        return parts[0], parts[1], true
}

// RT Cookie 名（__Host- prefix で強めの制約）
const rtCookieName = "__Host-rt"

// RT Cookie 有効期限（30日、秒）
const rtCookieMaxAge = 30 * 24 * 3600

// TokenHandler はトークンエンドポイントのハンドラ。
type TokenHandler struct {
        tokenUseCase   token.Interactor
        clientRepo     port.ClientRepository
        allowedOrigins []string // CSRF 用（refresh_token の Origin チェック）
}

// NewTokenHandler は TokenHandler を生成する。
func NewTokenHandler(tokenUseCase token.Interactor, clientRepo port.ClientRepository, allowedOrigins []string) *TokenHandler {
        return &TokenHandler{
                tokenUseCase:   tokenUseCase,
                clientRepo:     clientRepo,
                allowedOrigins: allowedOrigins,
        }
}

// Post は認可コードまたはリフレッシュトークンをトークンに交換する。
// POST /oauth/token
func (h *TokenHandler) Post(c echo.Context) error {
        var grantReq struct {
                GrantType string `form:"grant_type"`
        }
        if err := c.Bind(&grantReq); err != nil {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }
        switch grantReq.GrantType {
        case "authorization_code":
                return h.handleAuthorizationCode(c)
        case "refresh_token":
                return h.handleRefreshToken(c)
        default:
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "unsupported_grant_type"})
        }
}

func (h *TokenHandler) handleAuthorizationCode(c echo.Context) error {
        var req request.AuthCodeTokenRequest
        if err := c.Bind(&req); err != nil {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }
        if req.Code == "" || req.RedirectURI == "" || req.ClientID == "" {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }
        // PKCE S256 only: code_verifier 必須
        if req.CodeVerifier == "" {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }

        clientID, err := uuid.Parse(req.ClientID)
        if err != nil {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }

        // Confidential client: Authorization: Basic で client_secret が渡された場合に検証
        if clientIDStr, clientSecret, ok := parseBasicAuth(c.Request().Header.Get("Authorization")); ok && clientSecret != "" {
                parsedID, parseErr := uuid.Parse(clientIDStr)
                if parseErr != nil || parsedID != clientID {
                        c.Response().Header().Set("WWW-Authenticate", "Bearer")
                        return c.JSON(http.StatusUnauthorized, response.ErrorResponse{Error: "invalid_client"})
                }
                valid, _ := h.clientRepo.ValidateClientCredentials(c.Request().Context(), clientID, clientSecret)
                if !valid {
                        c.Response().Header().Set("WWW-Authenticate", "Bearer")
                        return c.JSON(http.StatusUnauthorized, response.ErrorResponse{Error: "invalid_client"})
                }
        }

        out, err := h.tokenUseCase.Exchange(c.Request().Context(), &token.ExchangeInput{
                Code:         req.Code,
                RedirectURI:  req.RedirectURI,
                ClientID:     clientID,
                CodeVerifier: req.CodeVerifier,
        })
        if err != nil {
                switch err {
                case token.ErrInvalidClient:
                        c.Response().Header().Set("WWW-Authenticate", "Bearer")
                        return c.JSON(http.StatusUnauthorized, response.ErrorResponse{Error: "invalid_client"})
                case token.ErrInvalidCode, token.ErrInvalidRedirectURI, token.ErrInvalidPKCE:
                        return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_grant"})
                default:
                        return c.JSON(http.StatusInternalServerError, response.ErrorResponse{Error: "server_error"})
                }
        }

        // refresh_token を Cookie で返却（body には含めない）
        setRefreshTokenCookie(c, out.RefreshToken)

        return c.JSON(http.StatusOK, response.TokenResponse{
                AccessToken: out.AccessToken,
                TokenType:   out.TokenType,
                ExpiresIn:   out.ExpiresIn,
                Scope:       out.Scope,
                IDToken:     out.IDToken,
        })
}

func (h *TokenHandler) handleRefreshToken(c echo.Context) error {
        // CSRF 対策: Origin / Referer が許可リストに含まれること（Cookie 送信のため）
        if !h.checkOrigin(c) {
                return c.JSON(http.StatusForbidden, response.ErrorResponse{Error: "invalid_request"})
        }

        // refresh_token は Cookie から取得（body には含めない）
        refreshToken := ""
        if cookie, err := c.Cookie(rtCookieName); err == nil && cookie != nil {
                refreshToken = cookie.Value
        }
        if refreshToken == "" {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }

        var req request.RefreshTokenRequest
        if err := c.Bind(&req); err != nil {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }
        if req.ClientID == "" {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }

        clientID, err := uuid.Parse(req.ClientID)
        if err != nil {
                return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_request"})
        }

        out, err := h.tokenUseCase.Refresh(c.Request().Context(), &token.RefreshInput{
                RefreshToken: refreshToken,
                ClientID:     clientID,
        })
        if err != nil {
                switch err {
                case token.ErrInvalidRefreshToken, token.ErrTokenExpired:
                        return c.JSON(http.StatusBadRequest, response.ErrorResponse{Error: "invalid_grant"})
                default:
                        return c.JSON(http.StatusInternalServerError, response.ErrorResponse{Error: "server_error"})
                }
        }

        // 新 RT を Cookie で返却（rotation 後）
        setRefreshTokenCookie(c, out.RefreshToken)

        return c.JSON(http.StatusOK, response.TokenResponse{
                AccessToken: out.AccessToken,
                TokenType:   out.TokenType,
                ExpiresIn:   out.ExpiresIn,
                Scope:       out.Scope,
                IDToken:     out.IDToken,
        })
}

// checkOrigin は Origin または Referer が許可リストに含まれるか確認する（CSRF 対策）。
func (h *TokenHandler) checkOrigin(c echo.Context) bool {
        origin := c.Request().Header.Get("Origin")
        if origin != "" {
                for _, allowed := range h.allowedOrigins {
                        if origin == allowed {
                                return true
                        }
                }
        }
        referer := c.Request().Header.Get("Referer")
        if referer != "" {
                for _, allowed := range h.allowedOrigins {
                        if strings.HasPrefix(referer, allowed+"/") || referer == allowed {
                                return true
                        }
                }
        }
        // 同一オリジン（Origin なし＝同一オリジンリクエスト）は許可
        if origin == "" && referer == "" {
                return true
        }
        return false
}

// setRefreshTokenCookie は refresh_token を HttpOnly Cookie で設定する。
func setRefreshTokenCookie(c echo.Context, rt string) {
        cookie := &http.Cookie{
                Name:     rtCookieName,
                Value:    rt,
                Path:     "/",
                MaxAge:   rtCookieMaxAge,
                HttpOnly: true,
                Secure:   true,
                SameSite: http.SameSiteLaxMode,
        }
        c.SetCookie(cookie)
}

        if auth == "" || !strings.HasPrefix(auth, "Bearer ") {
                return c.JSON(http.StatusUnauthorized, map[string]string{
                        "error": "invalid_token",
                })
        }
        accessToken := strings.TrimPrefix(auth, "Bearer ")

        out, err := h.userInfoUseCase.Get(c.Request().Context(), &userinfo.GetInput{
                AccessToken: accessToken,
        })
        if err != nil {
                if err == userinfo.ErrInvalidToken {
                        return c.JSON(http.StatusUnauthorized, map[string]string{
                                "error": "invalid_token",
                        })
                }
                return c.JSON(http.StatusInternalServerError, map[string]string{
                        "error": "server_error",
                })
        }

        return c.JSON(http.StatusOK, out)
}

                return nil, fmt.Errorf("ping database: %w", err)
        }
        return db, nil

import (

const defaultKeyID = "default"

// accessTokenClaims は JWT 内部の claims 構造。
// sub/aud は RegisteredClaims に含まれるため重複しない。
type accessTokenClaims struct {
        jwt.RegisteredClaims
        Sid string `json:"sid"`
        Ver int64  `json:"ver"`
}

// Signer は RS256 で JWT を署名する。port.JWTSigner と port.JWKSProvider を実装する。
type Signer struct {
        privateKey *rsa.PrivateKey
}

// NewSigner は秘密鍵から Signer を生成する。
func NewSigner(privateKey *rsa.PrivateKey) *Signer {
        return &Signer{privateKey: privateKey}
}

// GenerateKey は開発用の RSA 秘密鍵を生成する。
func GenerateKey() (*rsa.PrivateKey, error) {
        return rsa.GenerateKey(rand.Reader, 2048)
}

// ParsePrivateKey は PEM 形式の秘密鍵をパースする。
func ParsePrivateKey(pemBytes []byte) (*rsa.PrivateKey, error) {
        block, _ := pem.Decode(pemBytes)
        if block == nil {
                return nil, fmt.Errorf("failed to parse PEM")
        }
        key, err := x509.ParsePKCS1PrivateKey(block.Bytes)
        if err != nil {
                return nil, fmt.Errorf("parse private key: %w", err)
        }
        return key, nil
}

// idTokenClaims は ID Token の claims 構造。
type idTokenClaims struct {
        jwt.RegisteredClaims
        AuthTime *jwt.NumericDate `json:"auth_time"`
        Nonce   string            `json:"nonce,omitempty"`
}

// Sign はアクセストークン JWT を署名して返す。
func (s *Signer) Sign(sub, sid, aud uuid.UUID, ver int64, jti string, expiresIn time.Duration) (string, error) {
        now := time.Now()
        claims := accessTokenClaims{
                RegisteredClaims: jwt.RegisteredClaims{
                        ID:        jti,
                        Subject:   sub.String(),
                        Audience:  jwt.ClaimStrings{aud.String()},
                        ExpiresAt: jwt.NewNumericDate(now.Add(expiresIn)),
                        IssuedAt:  jwt.NewNumericDate(now),
                },
                Sid: sid.String(),
                Ver: ver,
        }

        token := jwt.NewWithClaims(jwt.SigningMethodRS256, claims)
        token.Header["kid"] = defaultKeyID
        return token.SignedString(s.privateKey)
}

// SignIDToken は ID Token JWT を署名して返す。
// iss, sub, aud, exp, iat, auth_time, nonce を含む。
func (s *Signer) SignIDToken(iss, sub, aud string, nonce string, authTime time.Time, expiresIn time.Duration) (string, error) {
        now := time.Now()
        claims := idTokenClaims{
                RegisteredClaims: jwt.RegisteredClaims{
                        Issuer:    iss,
                        Subject:   sub,
                        Audience:  jwt.ClaimStrings{aud},
                        ExpiresAt: jwt.NewNumericDate(now.Add(expiresIn)),
                        IssuedAt:  jwt.NewNumericDate(now),
                },
                AuthTime: jwt.NewNumericDate(authTime),
                Nonce:    nonce,
        }
        token := jwt.NewWithClaims(jwt.SigningMethodRS256, claims)
        token.Header["kid"] = defaultKeyID
        return token.SignedString(s.privateKey)
}

// JWKS は JSON Web Key Set の構造。
type JWKS struct {
        Keys []JWK `json:"keys"`
}

// JWK は JSON Web Key の構造（RSA 用）。
type JWK struct {
        Kty string `json:"kty"`
        Kid string `json:"kid"`
        Use string `json:"use"`
        Alg string `json:"alg"`
        N   string `json:"n"`
        E   string `json:"e"`
}

// ExportJWKS は公開鍵を JWKS 形式で返す。port.JWKSProvider を実装する。
func (s *Signer) ExportJWKS() any {
        return s.exportJWKS()
}

func (s *Signer) exportJWKS() *JWKS {
        pub := s.privateKey.Public().(*rsa.PublicKey)
        n := base64.RawURLEncoding.EncodeToString(pub.N.Bytes())
        e := base64.RawURLEncoding.EncodeToString(big.NewInt(int64(pub.E)).Bytes())
        return &JWKS{
                Keys: []JWK{
                        {Kty: "RSA", Kid: defaultKeyID, Use: "sig", Alg: "RS256", N: n, E: e},
                },
        }
}

// ExportJWKSJSON は JWKS を JSON 文字列で返す。
func (s *Signer) ExportJWKSJSON() (string, error) {
        b, err := json.Marshal(s.exportJWKS())
        if err != nil {
                return "", err
        }
        return string(b), nil
}

// Verify は JWT を検証し claims を返す。
func (s *Signer) Verify(tokenString string) (*port.AccessTokenClaims, error) {
        token, err := jwt.ParseWithClaims(tokenString, &accessTokenClaims{}, func(t *jwt.Token) (interface{}, error) {
                if _, ok := t.Method.(*jwt.SigningMethodRSA); !ok {
                        return nil, fmt.Errorf("unexpected signing method: %v", t.Header["alg"])
                }
                return &s.privateKey.PublicKey, nil
        })
        if err != nil {
                return nil, err
        }
        claims, ok := token.Claims.(*accessTokenClaims)
        if !ok || !token.Valid {
                return nil, fmt.Errorf("invalid token")
        }
        aud, _ := claims.GetAudience()
        audStr := ""
        if len(aud) > 0 {
                audStr = aud[0]
        }
        exp := int64(0)
        if claims.ExpiresAt != nil {
                exp = claims.ExpiresAt.Unix()
        }
        iat := int64(0)
        if claims.IssuedAt != nil {
                iat = claims.IssuedAt.Unix()
        }
        return &port.AccessTokenClaims{
                Sub: claims.Subject,
                Sid: claims.Sid,
                Ver: claims.Ver,
                Aud: audStr,
                Jti: claims.ID,
                Exp: exp,
                Iat: iat,
        }, nil
}

        var count int64
        err := r.db.WithContext(ctx).Model(&models.ClientModel{}).Where("id = ?", clientID).Count(&count).Error
        if err != nil {

        }
        return count > 0, nil

// ValidateRedirectURI は redirect_uri がクライアントに登録されているか検証する。
func (r *ClientRepository) ValidateRedirectURI(ctx context.Context, clientID uuid.UUID, redirectURI string) (bool, error) {
        var m models.ClientModel
        err := r.db.WithContext(ctx).Where("id = ?", clientID).First(&m).Error
        if err != nil {
                if errors.Is(err, gorm.ErrRecordNotFound) {
                        return false, nil
                }
                return false, err
        }
        uris := strings.FieldsFunc(m.RedirectURIs, func(r rune) bool {
                return r == ',' || r == '\n' || r == ' '
        })
        for _, uri := range uris {
                if strings.TrimSpace(uri) == redirectURI {
                        return true, nil
                }
        }
        return false, nil
}

// dummyBcryptHash は client が存在しない場合のタイミング攻撃対策用。
// 存在しない client でも bcrypt を1回回し、応答時間差で client の存在を推測されないようにする。
// bcrypt.GenerateFromPassword([]byte("dummy-timing-attack"), bcrypt.DefaultCost) で生成した固定ハッシュ
const dummyBcryptHash = "$2a$10$cZPGFEuZkd1TEWgZ1GFAmuqNTygVRNE.huBdcioIXAD2o5oTkQp9S"

// ValidateClientCredentials は client_id + client_secret でクライアント認証する。
// DB には client_secret_hash（bcrypt）を保存し、bcrypt.CompareHashAndPassword で照合する。
func (r *ClientRepository) ValidateClientCredentials(ctx context.Context, clientID uuid.UUID, clientSecret string) (bool, error) {
        if clientSecret == "" {
                return false, nil
        }
        var m models.ClientModel
        err := r.db.WithContext(ctx).Where("id = ?", clientID).First(&m).Error
        if err != nil {
                if errors.Is(err, gorm.ErrRecordNotFound) {
                        _ = bcrypt.CompareHashAndPassword([]byte(dummyBcryptHash), []byte(clientSecret))
                        return false, nil
                }
                return false, err
        }
        if m.ClientSecretHash == "" {
                return false, nil
        }
        err = bcrypt.CompareHashAndPassword([]byte(m.ClientSecretHash), []byte(clientSecret))
        return err == nil, nil
}

// GetAllowedScopes はクライアントが許可する scope の一覧を返す。
func (r *ClientRepository) GetAllowedScopes(ctx context.Context, clientID uuid.UUID) ([]string, error) {
        var m models.ClientModel
        err := r.db.WithContext(ctx).Where("id = ?", clientID).First(&m).Error
        if err != nil {
                if errors.Is(err, gorm.ErrRecordNotFound) {
                        return nil, nil
                }
                return nil, err
        }
        if m.AllowedScopes == "" {
                return []string{"openid", "profile", "email"}, nil
        }
        scopes := strings.FieldsFunc(m.AllowedScopes, func(r rune) bool {
                return r == ',' || r == '\n' || r == ' ' || r == '\t'
        })
        result := make([]string, 0, len(scopes))
        for _, s := range scopes {
                if t := strings.TrimSpace(s); t != "" {
                        result = append(result, t)
                }
        }
        if len(result) == 0 {
                return []string{"openid", "profile", "email"}, nil
        }
        return result, nil
}

// 認可コードの有効期限（10分）
const codeExpiryMinutes = 10

var loginChallengeTTL = 10 * time.Minute

var (
        ErrInvalidRedirectURI    = errors.New("invalid redirect_uri")
        ErrClientNotFound        = errors.New("client not found")
        ErrUnsupportedPKCE       = errors.New("PKCE: only S256 is supported")
        ErrInvalidResponseType   = errors.New("response_type must be code")
        ErrCodeChallengeRequired = errors.New("code_challenge is required")
        ErrStateRequired         = errors.New("state is required")
        ErrInvalidScope          = errors.New("scope is invalid or not allowed for client")
        ErrChallengeNotFound     = errors.New("login challenge not found or expired")
        ErrSessionRequired       = errors.New("session required")
)

// UseCase は認可コード発行のユースケース。

type UseCase struct {
        authCodeRepo port.AuthCodeRepository
        clientRepo   port.ClientRepository
        authReqRepo  port.AuthorizationRequestRepository
}

// NewUseCase は UseCase を生成する。
// セッションは触らない。sessionRepo は不要。
func NewUseCase(
        authCodeRepo port.AuthCodeRepository,
        clientRepo port.ClientRepository,
        authReqRepo port.AuthorizationRequestRepository,
) *UseCase {
        return &UseCase{
                authCodeRepo: authCodeRepo,
                clientRepo:   clientRepo,
                authReqRepo:  authReqRepo,
        }
}

// ValidateAuthorizeRequest は認可リクエストのパラメータを検証する。
func (u *UseCase) ValidateAuthorizeRequest(ctx context.Context, in *ValidateAuthorizeInput) error {
        if in.RedirectURI == "" || in.ClientID == uuid.Nil {
                return ErrInvalidRedirectURI
        }
        valid, err := u.clientRepo.ValidateRedirectURI(ctx, in.ClientID, in.RedirectURI)
        if err != nil {
                return fmt.Errorf("validate redirect_uri: %w", err)
        }
        if !valid {
                return ErrInvalidRedirectURI
        }
        if in.ResponseType != "code" {
                return ErrInvalidResponseType
        }
        if in.CodeChallenge == "" {
                return ErrCodeChallengeRequired
        }
        if in.CodeChallengeMethod != authcode.CodeChallengeMethodS256 {
                return ErrUnsupportedPKCE
        }
        if in.State == "" {
                return ErrStateRequired
        }
        scope := in.Scope
        if scope == "" {
                scope = "openid"
        }
        return u.validateScope(ctx, in.ClientID, scope)
}

// StoredAuthParams は CreateLoginChallenge / GetAndConsumeChallenge の入出力。
type StoredAuthParams struct {
        ClientID            string
        RedirectURI         string
        ResponseType        string
        Scope               string
        State               string
        Nonce               string
        CodeChallenge       string
        CodeChallengeMethod string
        Prompt              string // login, consent, none
        LoginURL            string // React ログイン画面の URL（CreateLoginChallenge 入力のみ）
}

// CreateLoginChallenge は未ログイン時に authorize リクエストを保存し、login_challenge を発行する。
func (u *UseCase) CreateLoginChallenge(ctx context.Context, params *StoredAuthParams) (challenge, redirectURL string, err error) {
        challenge, err = generateSecureCode()
        if err != nil {
                return "", "", fmt.Errorf("generate challenge: %w", err)
        }
        req := &port.StoredAuthorizationRequest{
                ClientID:            params.ClientID,
                RedirectURI:         params.RedirectURI,
                ResponseType:        params.ResponseType,
                Scope:               params.Scope,
                State:               params.State,
                Nonce:               params.Nonce,
                CodeChallenge:       params.CodeChallenge,
                CodeChallengeMethod: params.CodeChallengeMethod,
                Prompt:              params.Prompt,
        }
        if err := u.authReqRepo.Save(ctx, challenge, req, loginChallengeTTL); err != nil {
                return "", "", fmt.Errorf("save auth request: %w", err)
        }
        return challenge, params.LoginURL + "?login_challenge=" + challenge, nil
}

// GetChallenge は challenge に紐づく認可リクエストを取得する（消費しない）。

                State:               req.State,

// IssueCode は認可コードを発行する。
// セッションは /sessions/login で作成済み。SessionID を authcode に紐づけるだけ。セッションは触らない。
func (u *UseCase) IssueCode(ctx context.Context, in *IssueCodeInput) (*IssueCodeOutput, error) {
        // 1. redirect_uri の検証（client 登録済み URI と完全一致）
        valid, err := u.clientRepo.ValidateRedirectURI(ctx, in.ClientID, in.RedirectURI)
        if err != nil {
                return nil, fmt.Errorf("validate redirect_uri: %w", err)
        }
        if !valid {
                return nil, ErrInvalidRedirectURI
        }

        // 2. response_type=code 固定
        if in.ResponseType != "code" {
                return nil, ErrInvalidResponseType
        }

        // 3. code_challenge 必須、code_challenge_method=S256 固定（plain 禁止）
        if in.CodeChallenge == "" {
                return nil, ErrCodeChallengeRequired
        }
        if in.CodeChallengeMethod != authcode.CodeChallengeMethodS256 {
                return nil, ErrUnsupportedPKCE
        }

        // 4. state 必須
        if in.State == "" {
                return nil, ErrStateRequired
        }

        // 5. scope 検証（client 許可 scope 内か）
        if err := u.validateScope(ctx, in.ClientID, in.Scopes); err != nil {
                return nil, err
        }

        // 6. 認可コードの生成（平文を生成し、ハッシュをDBに保存）
        plainCode, err := generateSecureCode()
        if err != nil {
                return nil, fmt.Errorf("generate code: %w", err)
        }
        codeHash := hashCode(plainCode)

        // 7. 認可コードの作成と保存（session_id を紐付け。セッションは /sessions/login で作成済み）
        codeExpiresAt := time.Now().Add(codeExpiryMinutes * time.Minute)
        authCodeEntity := authcode.New(
                in.UserID,
                in.ClientID,
                in.SessionID,
                codeHash,
                in.CodeChallenge,
                in.CodeChallengeMethod,
                in.RedirectURI,
                in.Scopes,
                in.Nonce,
                codeExpiresAt,
        )

        if err := u.authCodeRepo.Save(ctx, authCodeEntity); err != nil {
                return nil, fmt.Errorf("save authorization code: %w", err)
        }

        return &IssueCodeOutput{
                Code:        plainCode,

// validateScope はリクエスト scope が client の許可 scope 内か検証する。
func (u *UseCase) validateScope(ctx context.Context, clientID uuid.UUID, requestedScope string) error {
        allowed, err := u.clientRepo.GetAllowedScopes(ctx, clientID)
        if err != nil {
                return fmt.Errorf("get allowed scopes: %w", err)
        }
        if len(allowed) == 0 {
                allowed = []string{"openid", "profile", "email"}
        }
        allowedSet := make(map[string]bool)
        for _, s := range allowed {
                allowedSet[s] = true
        }
        // リクエスト scope はスペース区切り
        requested := strings.Fields(requestedScope)
        for _, s := range requested {
                if !allowedSet[s] {
                        return ErrInvalidScope
                }
        }
        return nil
}

        if in.LoginChallenge == "" || in.Username == "" || in.Password == "" {

                in.IPAddress,
                in.UserAgent,
                1,
                expiresAt,
        )
        if err := u.sessionRepo.Save(ctx, sess); err != nil {
                return nil, fmt.Errorf("save session: %w", err)
        }

        nextURL := u.serverBaseURL + "/oauth/authorize/continue?login_challenge=" + in.LoginChallenge

        return &LoginOutput{
                NextURL:      nextURL,
                SessionToken: token,
        }, nil
}

// Exchange は認可コードをトークンに交換する。
// 1トランザクション内で code ロック→検証→消費→トークン発行を実行し、二重交換を防止する。
func (u *UseCase) Exchange(ctx context.Context, in *ExchangeInput) (*TokenOutput, error) {
        // A. grant_type は handler で検証済み

        // B. client 検証（最小: client_id 存在 + redirect_uri 一致）
        exists, err := u.clientRepo.Exists(ctx, in.ClientID)
        if err != nil {
                return nil, fmt.Errorf("client exists check: %w", err)
        }
        if !exists {
                return nil, ErrInvalidClient
        }
        ok, err := u.clientRepo.ValidateRedirectURI(ctx, in.ClientID, in.RedirectURI)
        if err != nil {
                return nil, fmt.Errorf("validate redirect uri: %w", err)
        }
        if !ok {
                return nil, ErrInvalidRedirectURI
        }

        var out *TokenOutput
        err = u.txRunner.Transaction(ctx, func(txCtx context.Context) error {
                // C. auth code 取得（ロック付き）
                codeHash := hashString(in.Code)
                authCodeEntity, err := u.authCodeRepo.FindByCodeHashForUpdate(txCtx, codeHash)
                if err != nil {
                        return fmt.Errorf("find auth code: %w", err)
                }
                if authCodeEntity == nil {
                        return ErrInvalidCode
                }

                // D. code が未使用
                if authCodeEntity.IsConsumed() {
                        return ErrInvalidCode
                }

                // E. code 期限切れ
                now := time.Now()
                if authCodeEntity.IsExpired(now) {
                        return ErrInvalidCode
                }

                // F. redirect_uri 一致（超重要）
                if authCodeEntity.RedirectURI() != in.RedirectURI {
                        return ErrInvalidRedirectURI
                }

                // G. client_id 一致
                if authCodeEntity.ClientID() != in.ClientID {
                        return ErrInvalidCode
                }

                // H. PKCE 検証（S256 only、code_challenge がある場合は code_verifier 必須）
                if authCodeEntity.CodeChallenge() != "" {
                        if in.CodeVerifier == "" {
                                return ErrInvalidPKCE
                        }
                        if authCodeEntity.CodeChallengeMethod() != authcode.CodeChallengeMethodS256 {
                                return ErrInvalidPKCE
                        }
                        expectedChallenge := computeCodeChallengeS256(in.CodeVerifier)
                        if expectedChallenge != authCodeEntity.CodeChallenge() {
                                return ErrInvalidPKCE
                        }
                }

                // I. code を使用済みに更新（トランザクション内）
                rowsAffected, err := u.authCodeRepo.MarkConsumed(txCtx, authCodeEntity.ID())
                if err != nil {
                        return fmt.Errorf("mark consumed: %w", err)
                }
                if rowsAffected != 1 {
                        return ErrInvalidCode
                }

                // J. トークン発行（session / token_version 取得、JWT、RT 保存）
                o, err := u.issueTokens(txCtx, authCodeEntity.UserID(), authCodeEntity.ClientID(), authCodeEntity.SessionID(), authCodeEntity.Scopes(), authCodeEntity.Nonce(), authCodeEntity.CreatedAt())
                if err != nil {
                        return err
                }
                out = o
                return nil
        })
        if err != nil {
                return nil, err
        }
        return out, nil
}

// RefreshInput はリフレッシュトークン交換の入力。
type RefreshInput struct {
        RefreshToken string
        ClientID     uuid.UUID
}

// Refresh はリフレッシュトークンで新しいトークンを発行する。
func (u *UseCase) Refresh(ctx context.Context, in *RefreshInput) (*TokenOutput, error) {
        // 1. リフレッシュトークンを取得
        tokenHash := hashString(in.RefreshToken)
        refreshToken, err := u.tokenRepo.FindRefreshTokenByHash(ctx, tokenHash)
        if err != nil || refreshToken == nil {
                return nil, ErrInvalidRefreshToken
        }

        if refreshToken.Revoked() {
                return nil, ErrInvalidRefreshToken
        }

        // 再利用検知（rotated 済み＝盗難の可能性）
        if refreshToken.Rotated() {
                accessToken, _ := u.tokenRepo.FindAccessTokenByID(ctx, refreshToken.AccessTokenID())
                if accessToken != nil && accessToken.SessionID() != uuid.Nil {
                        sid := accessToken.SessionID()
                        _ = u.sessionRepo.IncrementTokenVersion(ctx, sid)
                        _ = u.sessionRepo.Revoke(ctx, sid) // 疑わしい＝セッション強制失効
                }
                return nil, ErrInvalidRefreshToken
        }

        if time.Now().After(refreshToken.ExpiresAt()) {
                return nil, ErrTokenExpired
        }

        // 2. アクセストークンを取得して user_id, client_id, scopes を取得
        accessToken, err := u.tokenRepo.FindAccessTokenByID(ctx, refreshToken.AccessTokenID())
        if err != nil || accessToken == nil {
                return nil, ErrInvalidRefreshToken
        }

        // 3. client_id 一致
        if accessToken.ClientID() != in.ClientID {
                return nil, ErrInvalidRefreshToken
        }

        // 4. リフレッシュトークンをローテーション（旧トークン失効）
        if err := u.tokenRepo.MarkRefreshTokenRotated(ctx, refreshToken.ID()); err != nil {
                return nil, fmt.Errorf("mark refresh token rotated: %w", err)
        }

        // 5. 旧アクセストークン失効
        if err := u.tokenRepo.RevokeAccessToken(ctx, accessToken.ID()); err != nil {
                return nil, fmt.Errorf("revoke access token: %w", err)
        }

        // 6. 新トークン発行（session_id を引き継ぐ、refresh では nonce なし）
        return u.issueTokens(ctx, accessToken.UserID(), accessToken.ClientID(), accessToken.SessionID(), accessToken.Scopes(), "", time.Now())
}

// issueTokens はアクセストークン（RS256 JWT）、ID Token、リフレッシュトークンを発行する。
// nonce が空でない場合のみ ID Token に nonce claim を入れる。authTime は ID Token の auth_time。
func (u *UseCase) issueTokens(ctx context.Context, userID, clientID, sessionID uuid.UUID, scopes, nonce string, authTime time.Time) (*TokenOutput, error) {
        accessJTI, err := generateSecureToken()
        if err != nil {
                return nil, fmt.Errorf("generate access token: %w", err)
        }

        // token_version を取得（JWT claims に含める）
        sess, err := u.sessionRepo.FindByID(ctx, sessionID)
        if err != nil || sess == nil {
                return nil, fmt.Errorf("session not found: %w", err)
        }

        expiresIn := accessTokenExpirySeconds * time.Second
        jwtString, err := u.jwtSigner.Sign(userID, sessionID, clientID, sess.TokenVersion(), accessJTI, expiresIn)
        if err != nil {
                return nil, fmt.Errorf("sign jwt: %w", err)
        }

        // ID Token 発行（openid scope 想定、nonce は authcode から取得）
        var idToken string
        if u.issuer != "" {
                idToken, err = u.jwtSigner.SignIDToken(u.issuer, userID.String(), clientID.String(), nonce, authTime, idTokenExpirySeconds*time.Second)
                if err != nil {
                        return nil, fmt.Errorf("sign id token: %w", err)
                }
        }

        expiresAt := time.Now().Add(expiresIn)
        accessTokenID, err := u.tokenRepo.SaveAccessToken(ctx, userID, clientID, sessionID, accessJTI, scopes, expiresAt)
        if err != nil {
                return nil, fmt.Errorf("save access token: %w", err)
        }

        refreshToken, err := generateSecureToken()
        if err != nil {
                return nil, fmt.Errorf("generate refresh token: %w", err)
        }
        refreshHash := hashString(refreshToken)

        refreshExpiresAt := time.Now().Add(refreshTokenExpiryDays * 24 * time.Hour)
        _, err = u.tokenRepo.SaveRefreshToken(ctx, accessTokenID, refreshHash, refreshExpiresAt)
        if err != nil {
                return nil, fmt.Errorf("save refresh token: %w", err)
        }

        return &TokenOutput{
                AccessToken:  jwtString,
                TokenType:    "Bearer",
                ExpiresIn:    accessTokenExpirySeconds,
                RefreshToken: refreshToken,
                Scope:        scopes,
                IDToken:      idToken,
        }, nil
}

func hashString(s string) string {

        return hex.EncodeToString(h[:])
}

func computeCodeChallengeS256(verifier string) string {

        h := sha256.Sum256([]byte(verifier))
        return base64.URLEncoding.WithPadding(base64.NoPadding).EncodeToString(h[:])
}

// GetInput はユーザー情報取得の入力。

        AccessToken string
}

// GetOutput はユーザー情報の出力（OIDC UserInfo 形式）。
type GetOutput struct {
        Sub            string `json:"sub"`
        Email          string `json:"email,omitempty"`
        EmailVerified  bool   `json:"email_verified,omitempty"`
        PreferredUsername string `json:"preferred_username,omitempty"`
}

// Get はアクセストークン（JWT）からユーザー情報を取得する。
func (u *UseCase) Get(ctx context.Context, in *GetInput) (*GetOutput, error) {
        // 1. JWT 検証（署名・有効期限）
        claims, err := u.jwtSigner.Verify(in.AccessToken)
        if err != nil || claims == nil {
                return nil, ErrInvalidToken
        }

        // 2. access_token の失効チェック（jti で検索）
        accessToken, err := u.tokenRepo.FindAccessTokenByJTI(ctx, claims.Jti)
        if err != nil || accessToken == nil {
                return nil, ErrInvalidToken
        }
        if accessToken.Revoked() {
                return nil, ErrInvalidToken
        }

        // 3. Token Version 照合（sid → session → ver）
        sid, err := uuid.Parse(claims.Sid)
        if err != nil {
                return nil, ErrInvalidToken
        }
        sess, err := u.sessionRepo.FindByID(ctx, sid)
        if err != nil || sess == nil {
                return nil, ErrInvalidToken
        }
        if sess.TokenVersion() != claims.Ver {
                return nil, ErrInvalidToken
        }

        // 4. ユーザー情報取得
        userID, err := uuid.Parse(claims.Sub)
        if err != nil {
                return nil, ErrInvalidToken
        }
        usr, err := u.userRepo.FindByID(ctx, userID)
        if err != nil || usr == nil {
                return nil, ErrInvalidToken
        }

        return &GetOutput{
                Sub:               usr.ID().String(),
                Email:             usr.Email(),
                EmailVerified:     usr.EmailVerified(),
                PreferredUsername: usr.Username(),
        }, nil
}
